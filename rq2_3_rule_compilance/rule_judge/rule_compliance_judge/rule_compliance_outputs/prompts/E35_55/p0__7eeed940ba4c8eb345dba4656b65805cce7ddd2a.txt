+### useTheme Hook 设计规范
+```typescript
+// ✅ 正确的Hook设计
+export function useTheme() {
+  const [theme, setTheme] = useState<ThemeMode>('system');
+  const [isDark, setIsDark] = useState(false);
+
+  // 初始化主题
+  useEffect(() => {
+    initializeTheme();
+  }, []); // 只在组件挂载时执行
+
+  // 监听系统主题变化
+  useEffect(() => {
+    if (theme === 'system') {
+      const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)');
+      const handleChange = (e: MediaQueryListEvent) => {
+        setIsDark(e.matches);
+        applyTheme(e.matches ? 'dark' : 'light');
+      };
+      
+      mediaQuery.addEventListener('change', handleChange);
+      return () => mediaQuery.removeEventListener('change', handleChange);
+    }
+  }, [theme]); // 只依赖theme变化
+
+  return { theme, isDark, changeTheme };
+}
+```

+import { useState, useEffect } from 'react';
+import { ThemeMode } from '../types/electron.d';
+
+export function useTheme() {
+  const [theme, setTheme] = useState<ThemeMode>('system');
+  const [resolvedTheme, setResolvedTheme] = useState<'light' | 'dark'>('light');
+
+  // 获取系统主题
+  const getSystemTheme = (): 'light' | 'dark' => {
+    return window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light';
+  };
+
+  // 解析实际应用的主题
+  const resolveTheme = (themeMode: ThemeMode): 'light' | 'dark' => {
+    if (themeMode === 'system') {
+      return getSystemTheme();
+    }
+    return themeMode;
+  };
+
+  // 应用主题到DOM
+  const applyTheme = (resolvedTheme: 'light' | 'dark') => {
+    const root = document.documentElement;
+    root.classList.remove('light', 'dark');
+    root.classList.add(resolvedTheme);
+    
+    // 更新meta标签以支持系统标题栏
+    const metaTheme = document.querySelector('meta[name="theme-color"]');
+    if (metaTheme) {
+      metaTheme.setAttribute('content', resolvedTheme === 'dark' ? '#1f2937' : '#ffffff');
+    }
+  };
+
+  // 初始化主题
+  const initializeTheme = async () => {
+    try {
+      const config = await window.electronAPI.getAppConfig();
+      const themeMode = config.theme || 'system';
+      setTheme(themeMode);
+      
+      const resolved = resolveTheme(themeMode);
+      setResolvedTheme(resolved);
+      applyTheme(resolved);
+    } catch (error) {
+      console.error('初始化主题失败:', error);
+      // 降级到系统主题
+      const systemTheme = getSystemTheme();
+      setResolvedTheme(systemTheme);
+      applyTheme(systemTheme);
+    }
+  };
+
+  // 切换主题
+  const changeTheme = async (newTheme: ThemeMode) => {
+    try {
+      // 更新配置
+      await window.electronAPI.updateAppConfig({ theme: newTheme });
+      
+      // 更新本地状态
+      setTheme(newTheme);
+      
+      const resolved = resolveTheme(newTheme);
+      setResolvedTheme(resolved);
+      applyTheme(resolved);
+    } catch (error) {
+      console.error('切换主题失败:', error);
+    }
+  };
+
+  // 初始化主题（只在组件挂载时执行一次）
+  useEffect(() => {
+    initializeTheme();
+  }, []);
+
+  // 监听系统主题变化
+  useEffect(() => {
+    const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)');
+    
+    const handleSystemThemeChange = () => {
+      if (theme === 'system') {
+        const systemTheme = getSystemTheme();
+        setResolvedTheme(systemTheme);
+        applyTheme(systemTheme);
+      }
+    };
+
+    // 添加监听器
+    mediaQuery.addEventListener('change', handleSystemThemeChange);
+
+    // 清理监听器
+    return () => {
+      mediaQuery.removeEventListener('change', handleSystemThemeChange);
+    };
+  }, [theme]);
+
+  return {
+    theme,
+    resolvedTheme,
+    changeTheme,
+    isSystemTheme: theme === 'system'
+  };
