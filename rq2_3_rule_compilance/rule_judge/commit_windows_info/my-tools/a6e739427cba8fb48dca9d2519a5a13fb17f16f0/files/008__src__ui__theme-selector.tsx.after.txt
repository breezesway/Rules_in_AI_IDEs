import React from 'react';
import { Sun, Moon, Monitor, Check } from 'lucide-react';
import { ThemeMode } from '../types/electron.d';

interface ThemeSelectorProps {
  currentTheme: ThemeMode;
  onThemeChange: (theme: ThemeMode) => void;
  disabled?: boolean;
  isChanging?: boolean;
}

export function ThemeSelector({ currentTheme, onThemeChange, disabled, isChanging }: ThemeSelectorProps) {
  const themes: { mode: ThemeMode; label: string; icon: React.ReactNode; description: string; color: string }[] = [
    {
      mode: 'light',
      label: '浅色模式',
      icon: <Sun className="h-5 w-5" />,
      description: '适合光线充足的环境',
      color: 'text-yellow-600 dark:text-yellow-400'
    },
    {
      mode: 'dark',
      label: '深色模式',
      icon: <Moon className="h-5 w-5" />,
      description: '适合弱光环境，保护视力',
      color: 'text-indigo-600 dark:text-indigo-400'
    },
    {
      mode: 'system',
      label: '跟随系统',
      icon: <Monitor className="h-5 w-5" />,
      description: '自动跟随系统主题设置',
      color: 'text-green-600 dark:text-green-400'
    }
  ];

  return (
    <div className="space-y-4">
      {themes.map((theme) => (
        <button
          key={theme.mode}
          onClick={() => onThemeChange(theme.mode)}
          disabled={disabled}
          className={`
            group w-full flex items-center justify-between p-4 rounded-2xl border-2 transition-all duration-300 transform
            ${currentTheme === theme.mode
              ? 'border-blue-500 bg-blue-50 dark:border-blue-400 dark:bg-blue-900/20 shadow-lg scale-[1.02]'
              : 'border-gray-200 hover:border-blue-300 hover:bg-blue-50/50 dark:border-gray-600 dark:hover:border-blue-500 dark:hover:bg-blue-900/10 hover:scale-[1.01] hover:shadow-md'
            }
            ${disabled || isChanging ? 'opacity-50 cursor-not-allowed' : 'cursor-pointer'}
            ${isChanging ? 'pointer-events-none' : ''}
          `}
        >
          <div className="flex items-center space-x-4">
            <div className={`
              flex items-center justify-center w-12 h-12 rounded-2xl transition-all duration-300
              ${currentTheme === theme.mode
                ? 'bg-gradient-to-br from-blue-500 to-blue-600 text-white shadow-lg'
                : `bg-gray-100 dark:bg-gray-700 ${theme.color} group-hover:bg-blue-100 dark:group-hover:bg-blue-900/30`
              }
            `}>
              {theme.icon}
            </div>
            <div className="text-left">
              <div className={`font-semibold text-lg transition-colors duration-300 ${
                currentTheme === theme.mode 
                  ? 'text-blue-900 dark:text-blue-100' 
                  : 'text-gray-900 dark:text-gray-100 group-hover:text-blue-900 dark:group-hover:text-blue-100'
              }`}>
                {theme.label}
              </div>
              <div className={`text-sm transition-colors duration-300 ${
                currentTheme === theme.mode
                  ? 'text-blue-700 dark:text-blue-300'
                  : 'text-gray-500 dark:text-gray-400 group-hover:text-blue-600 dark:group-hover:text-blue-400'
              }`}>
                {theme.description}
              </div>
            </div>
          </div>
          
          {currentTheme === theme.mode && (
            <div className="flex items-center justify-center w-8 h-8 rounded-full bg-gradient-to-r from-blue-500 to-blue-600 text-white shadow-lg animate-pulse">
              <Check className="h-4 w-4 font-bold" />
            </div>
          )}
        </button>
      ))}
    </div>
  );
}
