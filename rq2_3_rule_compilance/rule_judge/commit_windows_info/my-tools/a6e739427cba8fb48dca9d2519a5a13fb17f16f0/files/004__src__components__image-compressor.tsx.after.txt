import React, { useState } from 'react';
import { Image, FolderOpen, Settings, Eye } from 'lucide-react';
import { Modal } from '../ui/modal';
import { ImagePreview } from '../ui/image-preview';
import { formatFileSize } from '../lib/utils';
import { toast } from 'sonner';

interface ImageCompressorProps {
  isOpen: boolean;
  onClose: () => void;
}

interface ImageFile {
  name: string;
  path: string;
  size: number;
  status: 'ready' | 'processing' | 'done' | 'error';
}

interface PreviewData {
  originalPath: string;
  previewPath: string;
  originalBase64: string;
  compressedBase64: string;
  original: {
    size: number;
    width: number;
    height: number;
    format: string;
  };
  compressed: {
    size: number;
    width: number;
    height: number;
    format: string;
  };
  compressionRatio: number;
  sizeDifference: number;
}



export function ImageCompressor({ isOpen, onClose }: ImageCompressorProps) {
  const [files, setFiles] = useState<ImageFile[]>([]);
  const [compressionMethod, setCompressionMethod] = useState('quality'); // 'quality', 'size', 'dimension'
  const [qualityLevel, setQualityLevel] = useState(70);
  const [targetSize, setTargetSize] = useState(500);
  const [maxDimension, setMaxDimension] = useState(1920);
  const [outputFormat, setOutputFormat] = useState('original');
  const [namingSuffix, setNamingSuffix] = useState('_compressed');
  const [saveLocation, setSaveLocation] = useState('');
  const [isProcessing, setIsProcessing] = useState(false);
  
  // 预览相关状态
  const [showPreview, setShowPreview] = useState(false);
  const [previewData, setPreviewData] = useState<PreviewData | null>(null);
  const [isGeneratingPreview, setIsGeneratingPreview] = useState(false);
  const [selectedFileForPreview, setSelectedFileForPreview] = useState<ImageFile | null>(null);

  // 初始化保存位置 - 每次打开时都重新获取最新配置
  React.useEffect(() => {
    if (isOpen) {
      window.electronAPI.getAppConfig().then(config => {
        if (config.defaultSaveLocation) {
          setSaveLocation(config.defaultSaveLocation);
        } else {
          setSaveLocation(''); // 如果没有设置默认位置，清空当前设置
        }
      }).catch(err => {
        console.error('获取应用配置失败:', err);
      });
    }
  }, [isOpen]); // 移除 saveLocation 依赖，确保每次打开都重新获取

  const handleFileSelect = async () => {
    try {
      const result = await window.electronAPI.selectImages();
      if (result.success && result.files) {
        const imageFiles: ImageFile[] = result.files.map(file => ({
          name: file.name,
          path: file.path,
          size: file.size,
          status: 'ready' as const
        }));
        setFiles(prev => [...prev, ...imageFiles]);
      }
    } catch (error) {
      console.error('选择文件失败:', error);
      toast.error('选择文件失败，请重试');
    }
  };

  const removeFile = (index: number) => {
    setFiles(prev => prev.filter((_, i) => i !== index));
  };

  // 生成预览
  const handlePreview = async (file: ImageFile) => {
    if (isGeneratingPreview) return;
    
    setSelectedFileForPreview(file);
    setIsGeneratingPreview(true);
    setShowPreview(true);
    setPreviewData(null);

    try {
      const settings = {
        method: compressionMethod,
        quality: qualityLevel,
        targetSize,
        maxDimension,
        outputFormat
      };

      const result = await window.electronAPI.generateImagePreview(file.path, settings);
      
      if (result.success && result.preview) {
        setPreviewData(result.preview);
      } else {
        toast.error(result.error || '生成预览失败');
        setShowPreview(false);
      }
    } catch (error) {
      console.error('生成预览失败:', error);
      toast.error('生成预览失败，请重试');
      setShowPreview(false);
    } finally {
      setIsGeneratingPreview(false);
    }
  };

  // 关闭预览
  const handleClosePreview = async () => {
    if (previewData?.previewPath) {
      // 清理预览文件
      try {
        await window.electronAPI.cleanupImagePreview(previewData.previewPath);
      } catch (error) {
        console.error('清理预览文件失败:', error);
      }
    }
    
    setShowPreview(false);
    setPreviewData(null);
    setSelectedFileForPreview(null);
  };

  // 确认压缩（从预览界面）
  const handleConfirmFromPreview = async () => {
    if (!selectedFileForPreview) return;
    
    // 关闭预览
    await handleClosePreview();
    
    // 执行压缩
    await handleCompress();
  };



  const handleCompress = async () => {
    if (files.length === 0) return;

    setIsProcessing(true);

    try {
      const settings = {
        method: compressionMethod,
        quality: qualityLevel,
        targetSize,
        maxDimension,
        outputFormat,
        namingSuffix,
        saveLocation: saveLocation || undefined
      };

      const result = await window.electronAPI.compressImages(
        files.map(f => f.path),
        settings
      );

      if (result.success) {
        toast.success(`成功压缩 ${result.processedCount} 张图片`);
        setFiles([]);
        onClose();
      } else {
        toast.error(result.error || '压缩失败');
      }
    } catch (error) {
      console.error('压缩失败:', error);
      toast.error('压缩失败，请重试');
    } finally {
      setIsProcessing(false);
    }
  };



  const footerContent = (
    <div className="flex justify-end space-x-3">
      <button
        onClick={onClose}
        className="px-6 py-3 border-2 border-gray-300 dark:border-gray-600 bg-white dark:bg-gray-800 text-gray-700 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-700 hover:border-gray-400 dark:hover:border-gray-500 rounded-xl font-medium transition-all duration-200"
      >
        关闭
      </button>
      <button
        onClick={handleCompress}
        disabled={files.length === 0 || isProcessing}
        className="px-8 py-3 bg-gradient-to-r from-green-600 to-green-700 hover:from-green-700 hover:to-green-800 dark:from-green-500 dark:to-green-600 dark:hover:from-green-600 dark:hover:to-green-700 text-white rounded-xl font-semibold shadow-lg hover:shadow-xl transform hover:scale-[1.02] disabled:opacity-50 disabled:cursor-not-allowed disabled:transform-none transition-all duration-200"
      >
        {isProcessing ? '压缩中...' : '开始压缩'}
      </button>
    </div>
  );

  return (
    <>
      <ImagePreview
        isOpen={showPreview}
        onClose={handleClosePreview}
        preview={previewData}
        onConfirm={handleConfirmFromPreview}
        isGenerating={isGeneratingPreview}
      />
      
      <Modal isOpen={isOpen} onClose={onClose} title="图片压缩" footer={footerContent}>
      <div className="space-y-6">
        {/* 文件选择区域 - 与PDF合并保持一致的样式 */}
        <div className="border-2 border-dashed border-green-300 dark:border-green-600 bg-green-50/30 dark:bg-green-900/10 rounded-2xl p-12 text-center hover:border-green-400 dark:hover:border-green-500 hover:bg-green-50/50 dark:hover:bg-green-900/20 transition-all duration-300">
          <div className="flex flex-col items-center space-y-6">
            <div className="p-6 bg-gradient-to-br from-green-100 to-green-200 dark:from-green-900/50 dark:to-green-800/50 rounded-3xl shadow-lg">
              <Image className="h-12 w-12 text-green-600 dark:text-green-400" />
            </div>
            <div className="space-y-3">
              <h3 className="text-2xl font-bold text-gray-900 dark:text-gray-100">选择图片文件进行压缩</h3>
              <p className="text-gray-600 dark:text-gray-400 max-w-md">
                支持 JPG、PNG、WebP 格式，批量压缩处理
              </p>
            </div>
            <button
              onClick={handleFileSelect}
              disabled={isProcessing}
              className="bg-gradient-to-r from-green-600 to-green-700 hover:from-green-700 hover:to-green-800 dark:from-green-500 dark:to-green-600 dark:hover:from-green-600 dark:hover:to-green-700 text-white px-8 py-4 rounded-xl font-semibold text-lg shadow-lg hover:shadow-xl transform hover:scale-[1.02] transition-all duration-200 disabled:opacity-50 disabled:cursor-not-allowed disabled:transform-none"
            >
              <span className="flex items-center space-x-2">
                <Image className="h-5 w-5" />
                <span>选择图片文件</span>
              </span>
            </button>
          </div>
        </div>

        {/* 文件列表 - 与PDF合并保持一致的样式 */}
        {files.length > 0 && (
          <div className="border border-gray-200 dark:border-gray-700 rounded-2xl overflow-hidden">
            <div className="p-4 bg-gray-50 dark:bg-gray-800 border-b border-gray-200 dark:border-gray-700">
              <div className="flex justify-between items-center">
                <div className="font-medium text-gray-900 dark:text-gray-100">
                  已选择的文件 ({files.length}) - 总大小: {formatFileSize(files.reduce((sum, f) => sum + f.size, 0))}
                </div>
              </div>
            </div>
            <ul className="divide-y divide-gray-200 dark:divide-gray-700">
              {files.map((file, index) => (
                <li 
                  key={index} 
                  className="flex items-center justify-between p-4 hover:bg-gray-50 dark:hover:bg-gray-800 transition-all duration-200"
                >
                  <div className="flex items-center space-x-3">
                    <div className={`p-2 rounded-xl ${
                      file.status === 'ready' ? 'bg-green-100 dark:bg-green-900' :
                      file.status === 'processing' ? 'bg-yellow-100 dark:bg-yellow-900' :
                      file.status === 'done' ? 'bg-blue-100 dark:bg-blue-900' : 'bg-red-100 dark:bg-red-900'
                    }`}>
                      <Image className={`h-5 w-5 ${
                        file.status === 'ready' ? 'text-green-600 dark:text-green-400' :
                        file.status === 'processing' ? 'text-yellow-600 dark:text-yellow-400' :
                        file.status === 'done' ? 'text-blue-600 dark:text-blue-400' : 'text-red-600 dark:text-red-400'
                      }`} />
                    </div>
                    <div>
                      <p className="font-medium text-gray-900 dark:text-gray-100">{file.name}</p>
                      <div className="flex items-center space-x-2">
                        <p className="text-sm text-gray-500 dark:text-gray-400">
                          {formatFileSize(file.size)}
                        </p>
                        <span className={`text-xs px-2 py-1 rounded-full ${
                          file.status === 'ready' ? 'bg-green-100 text-green-700 dark:bg-green-900/30 dark:text-green-400' :
                          file.status === 'processing' ? 'bg-yellow-100 text-yellow-700 dark:bg-yellow-900/30 dark:text-yellow-400' :
                          file.status === 'done' ? 'bg-blue-100 text-blue-700 dark:bg-blue-900/30 dark:text-blue-400' : 
                          'bg-red-100 text-red-700 dark:bg-red-900/30 dark:text-red-400'
                        }`}>
                          {file.status === 'ready' ? '准备就绪' :
                           file.status === 'processing' ? '处理中' :
                           file.status === 'done' ? '已完成' : '错误'}
                        </span>
                      </div>
                    </div>
                  </div>
                  <div className="flex items-center space-x-2">
                    <button
                      onClick={() => handlePreview(file)}
                      disabled={isProcessing || isGeneratingPreview}
                      className="text-green-500 dark:text-green-400 hover:text-green-700 dark:hover:text-green-300 hover:bg-green-50 dark:hover:bg-green-900 p-2 rounded-xl transition-all duration-200 disabled:opacity-50 disabled:cursor-not-allowed"
                      title="预览压缩效果"
                    >
                      <Eye className="h-4 w-4" />
                    </button>
                    <button
                      onClick={() => removeFile(index)}
                      disabled={isProcessing}
                      className="text-red-500 dark:text-red-400 hover:text-red-700 dark:hover:text-red-300 hover:bg-red-50 dark:hover:bg-red-900 p-2 rounded-xl transition-all duration-200 disabled:opacity-50 disabled:cursor-not-allowed"
                    >
                      删除
                    </button>
                  </div>
                </li>
              ))}
            </ul>
          </div>
        )}

        {/* 压缩设置 - 改进分组样式 */}
        <div className="bg-gray-50 dark:bg-gray-800/50 rounded-2xl p-6 space-y-6">
          <div className="flex items-center space-x-3 mb-2">
            <div className="p-2 bg-green-100 dark:bg-green-900/50 rounded-xl">
              <Settings className="h-6 w-6 text-green-600 dark:text-green-400" />
            </div>
            <div>
              <h3 className="text-xl font-semibold text-gray-900 dark:text-gray-100">压缩设置</h3>
              <p className="text-sm text-gray-600 dark:text-gray-400">选择压缩方式和参数</p>
            </div>
          </div>

          {/* 压缩方式选择 - 改进样式 */}
          <div className="grid grid-cols-3 gap-4">
            <button
              onClick={() => setCompressionMethod('quality')}
              className={`group p-5 rounded-2xl border-2 text-center transition-all duration-300 transform ${
                compressionMethod === 'quality'
                  ? 'border-green-500 bg-green-50 dark:border-green-400 dark:bg-green-900/20 shadow-lg scale-[1.02]'
                  : 'border-gray-200 hover:border-green-300 hover:bg-green-50/50 dark:border-gray-600 dark:hover:border-green-500 dark:hover:bg-green-900/10 hover:scale-[1.01] hover:shadow-md'
              }`}
            >
              <div className={`font-semibold transition-colors duration-300 ${
                compressionMethod === 'quality' 
                  ? 'text-green-900 dark:text-green-100' 
                  : 'text-gray-900 dark:text-gray-100 group-hover:text-green-900 dark:group-hover:text-green-100'
              }`}>按质量等级</div>
              <div className={`text-sm mt-2 transition-colors duration-300 ${
                compressionMethod === 'quality'
                  ? 'text-green-700 dark:text-green-300'
                  : 'text-gray-500 dark:text-gray-400 group-hover:text-green-600 dark:group-hover:text-green-400'
              }`}>调整压缩质量</div>
            </button>
            <button
              onClick={() => setCompressionMethod('size')}
              className={`group p-5 rounded-2xl border-2 text-center transition-all duration-300 transform ${
                compressionMethod === 'size'
                  ? 'border-green-500 bg-green-50 dark:border-green-400 dark:bg-green-900/20 shadow-lg scale-[1.02]'
                  : 'border-gray-200 hover:border-green-300 hover:bg-green-50/50 dark:border-gray-600 dark:hover:border-green-500 dark:hover:bg-green-900/10 hover:scale-[1.01] hover:shadow-md'
              }`}
            >
              <div className={`font-semibold transition-colors duration-300 ${
                compressionMethod === 'size' 
                  ? 'text-green-900 dark:text-green-100' 
                  : 'text-gray-900 dark:text-gray-100 group-hover:text-green-900 dark:group-hover:text-green-100'
              }`}>按文件大小</div>
              <div className={`text-sm mt-2 transition-colors duration-300 ${
                compressionMethod === 'size'
                  ? 'text-green-700 dark:text-green-300'
                  : 'text-gray-500 dark:text-gray-400 group-hover:text-green-600 dark:group-hover:text-green-400'
              }`}>压缩至指定大小</div>
            </button>
            <button
              onClick={() => setCompressionMethod('dimension')}
              className={`group p-5 rounded-2xl border-2 text-center transition-all duration-300 transform ${
                compressionMethod === 'dimension'
                  ? 'border-green-500 bg-green-50 dark:border-green-400 dark:bg-green-900/20 shadow-lg scale-[1.02]'
                  : 'border-gray-200 hover:border-green-300 hover:bg-green-50/50 dark:border-gray-600 dark:hover:border-green-500 dark:hover:bg-green-900/10 hover:scale-[1.01] hover:shadow-md'
              }`}
            >
              <div className={`font-semibold transition-colors duration-300 ${
                compressionMethod === 'dimension' 
                  ? 'text-green-900 dark:text-green-100' 
                  : 'text-gray-900 dark:text-gray-100 group-hover:text-green-900 dark:group-hover:text-green-100'
              }`}>按图片尺寸</div>
              <div className={`text-sm mt-2 transition-colors duration-300 ${
                compressionMethod === 'dimension'
                  ? 'text-green-700 dark:text-green-300'
                  : 'text-gray-500 dark:text-gray-400 group-hover:text-green-600 dark:group-hover:text-green-400'
              }`}>调整图片尺寸</div>
            </button>
          </div>

          {/* 压缩参数设置 */}
          {compressionMethod === 'quality' && (
            <div>
              <label className="block text-sm font-medium mb-3 text-gray-900 dark:text-gray-100">
                质量等级: {qualityLevel}%
              </label>
              <div className="px-1">
                <input
                  type="range"
                  min="10"
                  max="100"
                  value={qualityLevel}
                  onChange={(e) => setQualityLevel(parseInt(e.target.value))}
                  className="w-full h-2 bg-gray-200 dark:bg-gray-600 rounded-xl appearance-none cursor-pointer slider"
                />
                <div className="flex justify-between text-xs text-gray-500 dark:text-gray-400 mt-2">
                  <span>高压缩</span>
                  <span>高质量</span>
                </div>
              </div>
            </div>
          )}

          {compressionMethod === 'size' && (
            <div>
              <label className="block text-sm font-medium mb-2 text-gray-900 dark:text-gray-100">
                目标文件大小 (KB)
              </label>
              <input
                type="number"
                value={targetSize}
                onChange={(e) => setTargetSize(parseInt(e.target.value))}
                className="w-full border border-gray-300 dark:border-gray-600 rounded-xl px-4 py-3 focus:ring-2 focus:ring-gray-900 dark:focus:ring-gray-100 focus:border-transparent transition-all duration-200 bg-white dark:bg-gray-800 text-gray-900 dark:text-gray-100"
                placeholder="500"
              />
              <p className="text-sm text-gray-500 dark:text-gray-400 mt-2">压缩后的图片将尽量接近此大小</p>
            </div>
          )}

          {compressionMethod === 'dimension' && (
            <div>
              <label className="block text-sm font-medium mb-2 text-gray-900 dark:text-gray-100">
                最大尺寸 (像素)
              </label>
              <input
                type="number"
                value={maxDimension}
                onChange={(e) => setMaxDimension(parseInt(e.target.value))}
                className="w-full border border-gray-300 dark:border-gray-600 rounded-xl px-4 py-3 focus:ring-2 focus:ring-gray-900 dark:focus:ring-gray-100 focus:border-transparent transition-all duration-200 bg-white dark:bg-gray-800 text-gray-900 dark:text-gray-100"
                placeholder="1920"
              />
              <p className="text-sm text-gray-500 dark:text-gray-400 mt-2">图片的宽度或高度将不超过此值，保持原始比例</p>
            </div>
          )}

          {/* 输出格式和文件名设置 */}
          <div className="grid grid-cols-2 gap-4">
            <div>
              <label className="block text-sm font-medium mb-2 text-gray-900 dark:text-gray-100">输出格式</label>
              <select
                value={outputFormat}
                onChange={(e) => setOutputFormat(e.target.value)}
                className="w-full border border-gray-300 dark:border-gray-600 rounded-xl px-4 py-3 focus:ring-2 focus:ring-gray-900 dark:focus:ring-gray-100 focus:border-transparent transition-all duration-200 bg-white dark:bg-gray-800 text-gray-900 dark:text-gray-100"
              >
                <option value="original">保持原格式</option>
                <option value="jpg">JPG</option>
                <option value="png">PNG</option>
                <option value="webp">WebP</option>
              </select>
            </div>
            <div>
              <label className="block text-sm font-medium mb-2 text-gray-900 dark:text-gray-100">文件名后缀</label>
              <input
                type="text"
                value={namingSuffix}
                onChange={(e) => setNamingSuffix(e.target.value)}
                className="w-full border border-gray-300 dark:border-gray-600 rounded-xl px-4 py-3 focus:ring-2 focus:ring-gray-900 dark:focus:ring-gray-100 focus:border-transparent transition-all duration-200 bg-white dark:bg-gray-800 text-gray-900 dark:text-gray-100"
                placeholder="_compressed"
              />
            </div>
          </div>

          {/* 保存位置信息 */}
          <div>
            <label className="block text-sm font-medium mb-2 text-gray-900 dark:text-gray-100">保存位置</label>
            <div className="p-3 bg-gray-50 dark:bg-gray-800 border border-gray-200 dark:border-gray-700 rounded-xl">
              {saveLocation ? (
                <p className="text-sm text-gray-700 dark:text-gray-300">将保存到: {saveLocation}</p>
              ) : (
                <p className="text-sm text-gray-700 dark:text-gray-300">将使用应用设置中配置的默认保存位置</p>
              )}
              <p className="text-xs text-gray-500 dark:text-gray-400 mt-1">
                可在应用设置中修改默认保存位置
              </p>
            </div>
          </div>
        </div>
      </div>
    </Modal>
    </>
  );
}
