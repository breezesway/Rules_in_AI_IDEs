import React, { useState, useEffect } from 'react';
import { RotateCcw, Save, FileText, FolderOpen } from 'lucide-react';
import { Modal } from '../ui/modal';
import { SaveLocationSelector } from '../ui/save-location-selector';
import { AppConfig } from '../types/electron.d';
import { toast } from 'sonner';

interface AppSettingsProps {
  isOpen: boolean;
  onClose: () => void;
}

export function AppSettings({ isOpen, onClose }: AppSettingsProps) {
  const [config, setConfig] = useState<AppConfig>({
    autoNumberDuplicateFiles: true,
    defaultSaveLocation: ''
  });
  const [loading, setLoading] = useState(false);
  const [saving, setSaving] = useState(false);

  // 加载配置
  const loadConfig = async () => {
    setLoading(true);
    try {
      const currentConfig = await window.electronAPI.getAppConfig();
      setConfig(currentConfig);
    } catch (error) {
      console.error('加载配置失败:', error);
      toast.error('加载配置失败');
    } finally {
      setLoading(false);
    }
  };

  // 保存配置
  const saveConfig = async () => {
    setSaving(true);
    try {
      const result = await window.electronAPI.updateAppConfig(config);
      if (result.success) {
        toast.success('配置已保存');
      } else {
        toast.error('保存失败: ' + result.error);
      }
    } catch (error) {
      console.error('保存配置失败:', error);
      toast.error('保存配置失败');
    } finally {
      setSaving(false);
    }
  };

  // 重置配置
  const resetConfig = async () => {
    if (!window.confirm('确定要重置所有配置为默认值吗？')) {
      return;
    }

    try {
      const result = await window.electronAPI.resetAppConfig();
      if (result.success) {
        await loadConfig(); // 重新加载配置
        toast.success('配置已重置为默认值');
      } else {
        toast.error('重置失败: ' + result.error);
      }
    } catch (error) {
      console.error('重置配置失败:', error);
      toast.error('重置配置失败');
    }
  };

  // 更新配置项
  const updateConfig = (key: keyof AppConfig, value: boolean | string) => {
    setConfig(prev => ({
      ...prev,
      [key]: value
    }));
  };

  // 选择默认保存位置
  const handleSelectDefaultSaveLocation = async () => {
    try {
      const result = await window.electronAPI.selectSaveLocation();
      if (result.success && result.path) {
        updateConfig('defaultSaveLocation', result.path);
      }
    } catch (error) {
      console.error('选择保存位置失败:', error);
      toast.error('选择保存位置失败，请重试');
    }
  };

  useEffect(() => {
    if (isOpen) {
      loadConfig();
    }
  }, [isOpen]);

  const footerContent = (
    <div className="flex justify-between">
      <button
        onClick={resetConfig}
        disabled={loading || saving}
        className="flex items-center space-x-2 px-4 py-3 border border-gray-300 text-gray-600 rounded-xl hover:bg-gray-50 font-medium transition-colors duration-200 disabled:opacity-50 disabled:cursor-not-allowed"
      >
        <RotateCcw className="h-4 w-4" />
        <span>重置为默认</span>
      </button>
      
      <div className="flex space-x-3">
        <button
          onClick={onClose}
          disabled={saving}
          className="px-6 py-3 border border-gray-300 rounded-xl hover:bg-gray-50 font-medium transition-colors duration-200 disabled:opacity-50"
        >
          取消
        </button>
        <button
          onClick={saveConfig}
          disabled={loading || saving}
          className="flex items-center space-x-2 px-6 py-3 bg-gray-900 text-white rounded-xl font-medium hover:bg-gray-800 disabled:opacity-50 disabled:cursor-not-allowed transition-colors duration-200"
        >
          {saving ? (
            <>
              <div className="animate-spin rounded-full h-4 w-4 border-b-2 border-white"></div>
              <span>保存中...</span>
            </>
          ) : (
            <>
              <Save className="h-4 w-4" />
              <span>保存配置</span>
            </>
          )}
        </button>
      </div>
    </div>
  );

  return (
    <Modal isOpen={isOpen} onClose={onClose} title="应用设置" footer={footerContent}>
      <div className="space-y-6">
        {loading ? (
          <div className="text-center py-8">
            <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-gray-900 mx-auto"></div>
            <p className="mt-2 text-gray-500">加载中...</p>
          </div>
        ) : (
          <div className="space-y-6">
            {/* 文件保存设置 */}
            <div className="space-y-4">
              <div className="flex items-center space-x-2 mb-4">
                <FolderOpen className="h-5 w-5 text-gray-600" />
                <h3 className="text-lg font-medium text-gray-900">文件保存</h3>
              </div>

              {/* 默认保存位置设置 */}
              <div className="p-4 border border-gray-200 rounded-xl">
                <div className="space-y-4">
                  <div>
                    <h4 className="font-medium text-gray-900 mb-2">
                      默认保存位置
                    </h4>
                    <p className="text-sm text-gray-600 mb-3">
                      设置所有工具的默认文件保存位置。如果不设置，将使用系统默认下载目录。
                    </p>
                  </div>
                  
                  <div>
                    <label className="block text-sm font-medium mb-2 text-gray-900">保存位置</label>
                    <SaveLocationSelector
                      value={config.defaultSaveLocation}
                      placeholder="使用系统默认下载目录"
                      onSelect={handleSelectDefaultSaveLocation}
                      disabled={loading || saving}
                    />
                    {config.defaultSaveLocation && (
                      <p className="text-sm text-gray-500 mt-2">当前设置: {config.defaultSaveLocation}</p>
                    )}
                  </div>
                </div>
              </div>
            </div>

            {/* 文件处理设置 */}
            <div className="space-y-4">
              <div className="flex items-center space-x-2 mb-4">
                <FileText className="h-5 w-5 text-gray-600" />
                <h3 className="text-lg font-medium text-gray-900">文件处理</h3>
              </div>

              {/* 自动添加数字序号设置 */}
              <div className="p-4 border border-gray-200 rounded-xl">
                <div className="flex items-start justify-between">
                  <div className="flex-1">
                    <h4 className="font-medium text-gray-900 mb-2">
                      文件名重复时自动添加序号
                    </h4>
                    <p className="text-sm text-gray-600 mb-3">
                      当输出文件名已存在时，自动在文件名后添加数字序号（如 document(1).pdf），
                      而不是询问是否覆盖原文件。
                    </p>
                    <div className="text-xs text-gray-500 space-y-1">
                      <p>• 启用：自动生成 document(1).pdf, document(2).pdf...</p>
                      <p>• 禁用：弹出对话框询问是否覆盖现有文件</p>
                    </div>
                  </div>
                  
                  <div className="ml-4 flex-shrink-0">
                    <label className="relative inline-flex items-center cursor-pointer">
                      <input
                        type="checkbox"
                        checked={config.autoNumberDuplicateFiles}
                        onChange={(e) => updateConfig('autoNumberDuplicateFiles', e.target.checked)}
                        className="sr-only peer"
                      />
                      <div className="w-11 h-6 bg-gray-200 peer-focus:outline-none peer-focus:ring-4 peer-focus:ring-gray-300 rounded-full peer peer-checked:after:translate-x-full peer-checked:after:border-white after:content-[''] after:absolute after:top-[2px] after:left-[2px] after:bg-white after:border-gray-300 after:border after:rounded-full after:h-5 after:w-5 after:transition-all peer-checked:bg-gray-900"></div>
                    </label>
                  </div>
                </div>
                
                {/* 示例展示 */}
                <div className="mt-4 p-3 bg-gray-50 rounded-lg">
                  <p className="text-sm font-medium text-gray-700 mb-2">示例：</p>
                  <div className="text-sm text-gray-600 space-y-1">
                    {config.autoNumberDuplicateFiles ? (
                      <>
                        <p>✓ 原文件：document.pdf</p>
                        <p>✓ 新文件：document(1).pdf（自动生成）</p>
                        <p>✓ 再次保存：document(2).pdf（自动生成）</p>
                      </>
                    ) : (
                      <>
                        <p>⚠ 原文件：document.pdf</p>
                        <p>⚠ 新文件：document.pdf（询问是否覆盖）</p>
                        <p>⚠ 需要用户选择：覆盖 或 取消</p>
                      </>
                    )}
                  </div>
                </div>
              </div>
            </div>

            {/* 配置信息 */}
            <div className="pt-4 border-t border-gray-200">
              <p className="text-xs text-gray-500">
                配置文件保存在应用数据目录中，重装应用后配置会保留。
              </p>
            </div>
          </div>
        )}
      </div>
    </Modal>
  );
}
