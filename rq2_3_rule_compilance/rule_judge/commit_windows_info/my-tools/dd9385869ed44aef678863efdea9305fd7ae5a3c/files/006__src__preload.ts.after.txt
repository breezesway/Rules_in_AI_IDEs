import { contextBridge, ipcRenderer } from 'electron';

// 暴露PDF合并相关的API给渲染进程
contextBridge.exposeInMainWorld('electronAPI', {
  // PDF合并
  mergePdfs: (options: {
    filePaths: string[];
    outputPath: string;
    outputFileName: string;
  }) => ipcRenderer.invoke('merge-pdfs', options),
  
  // 监听合并进度
  onMergeProgress: (callback: (progress: {
    current: number;
    total: number;
    currentFile: string;
    progress: number;
  }) => void) => {
    const listener = (_event: unknown, progress: {
      current: number;
      total: number;
      currentFile: string;
      progress: number;
    }) => callback(progress);
    ipcRenderer.on('merge-progress', listener);
    
    // 返回清理函数
    return () => {
      ipcRenderer.removeListener('merge-progress', listener);
    };
  },
  
  // 选择保存位置
  selectSavePath: () => ipcRenderer.invoke('select-save-path'),
  
  // 获取默认下载目录
  getDownloadsPath: () => ipcRenderer.invoke('get-downloads-path'),
  
  // 选择PDF文件
  selectPdfFiles: () => ipcRenderer.invoke('select-pdf-files'),
  
  openFileLocation: (filePath: string) => ipcRenderer.invoke('open-file-location', filePath),
  
  // 历史记录相关API
  getOperationHistory: () => ipcRenderer.invoke('get-operation-history'),
  clearOperationHistory: () => ipcRenderer.invoke('clear-operation-history'),
  deleteHistoryRecord: (id: string) => ipcRenderer.invoke('delete-history-record', id),
  
  // 配置相关API
  getAppConfig: () => ipcRenderer.invoke('get-app-config'),
  updateAppConfig: (config: Record<string, unknown>) => ipcRenderer.invoke('update-app-config', config),
  resetAppConfig: () => ipcRenderer.invoke('reset-app-config'),
  
  // 图片压缩相关API
  selectImages: () => ipcRenderer.invoke('select-images'),
  selectSaveLocation: () => ipcRenderer.invoke('select-save-location'),
  compressImages: (filePaths: string[], settings: Record<string, unknown>) => 
    ipcRenderer.invoke('compress-images', filePaths, settings),
  
  // 监听压缩进度
  onCompressionProgress: (callback: (progress: {
    currentFile: string;
    currentIndex: number;
    totalFiles: number;
    percentage: number;
  }) => void) => {
    const listener = (_event: unknown, progress: {
      currentFile: string;
      currentIndex: number;
      totalFiles: number;
      percentage: number;
    }) => callback(progress);
    ipcRenderer.on('compression-progress', listener);
    
    // 返回清理函数
    return () => {
      ipcRenderer.removeListener('compression-progress', listener);
    };
  },
  
  removeCompressionProgressListener: () => {
    ipcRenderer.removeAllListeners('compression-progress');
  },
});
