// 历史记录类型定义
export interface HistoryRecord {
  id: string;
  type: 'pdf-merge' | 'image-compress';
  timestamp: number;
  files: {
    name: string;
    size: number;
    path: string;
  }[];
  output: {
    name: string;
    path: string;
    size?: number;
  };
  settings: Record<string, unknown>;
  duration?: number;
  success: boolean;
  error?: string;
}

// 应用配置类型定义
export interface AppConfig {
  autoNumberDuplicateFiles: boolean;
  defaultSaveLocation: string;
}

// PDF合并选项类型
export interface PdfMergeOptions {
  filePaths: string[];
  outputPath: string;
  outputFileName: string;
}

// 图片压缩设置类型
export interface ImageCompressionSettings {
  method: string;
  quality?: number;
  targetSize?: number;
  maxDimension?: number;
  outputFormat?: string;
  namingSuffix?: string;
  saveLocation?: string;
}

// 图片预览设置类型
export interface ImagePreviewSettings {
  method: string;
  quality?: number;
  targetSize?: number;
  maxDimension?: number;
  outputFormat?: string;
}
