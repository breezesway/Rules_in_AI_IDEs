import React from 'react';
import { Sun, Moon, Monitor, Check } from 'lucide-react';
import { ThemeMode } from '../types/electron.d';

interface ThemeSelectorProps {
  currentTheme: ThemeMode;
  onThemeChange: (theme: ThemeMode) => void;
  disabled?: boolean;
  isChanging?: boolean;
}

export function ThemeSelector({ currentTheme, onThemeChange, disabled, isChanging }: ThemeSelectorProps) {
  const themes: { mode: ThemeMode; label: string; icon: React.ReactNode; description: string }[] = [
    {
      mode: 'light',
      label: '浅色模式',
      icon: <Sun className="h-4 w-4" />,
      description: '始终使用浅色主题'
    },
    {
      mode: 'dark',
      label: '深色模式',
      icon: <Moon className="h-4 w-4" />,
      description: '始终使用深色主题'
    },
    {
      mode: 'system',
      label: '跟随系统',
      icon: <Monitor className="h-4 w-4" />,
      description: '根据系统设置自动切换'
    }
  ];

  return (
    <div className="space-y-3">
      {themes.map((theme) => (
        <button
          key={theme.mode}
          onClick={() => onThemeChange(theme.mode)}
          disabled={disabled}
          className={`
            w-full flex items-center justify-between p-3 rounded-xl border transition-all duration-200
            ${currentTheme === theme.mode
              ? 'border-gray-900 bg-gray-50 dark:border-gray-100 dark:bg-gray-800'
              : 'border-gray-200 hover:border-gray-300 hover:bg-gray-50 dark:border-gray-700 dark:hover:border-gray-600 dark:hover:bg-gray-800'
            }
            ${disabled || isChanging ? 'opacity-50 cursor-not-allowed' : 'cursor-pointer'}
            ${isChanging ? 'pointer-events-none' : ''}
          `}
        >
          <div className="flex items-center space-x-3">
            <div className={`
              flex items-center justify-center w-8 h-8 rounded-lg
              ${currentTheme === theme.mode
                ? 'bg-gray-900 text-white dark:bg-gray-100 dark:text-gray-900'
                : 'bg-gray-100 text-gray-600 dark:bg-gray-700 dark:text-gray-300'
              }
            `}>
              {theme.icon}
            </div>
            <div className="text-left">
              <div className="font-medium text-gray-900 dark:text-gray-100">
                {theme.label}
              </div>
              <div className="text-sm text-gray-500 dark:text-gray-400">
                {theme.description}
              </div>
            </div>
          </div>
          
          {currentTheme === theme.mode && (
            <div className="flex items-center justify-center w-5 h-5 rounded-full bg-gray-900 text-white dark:bg-gray-100 dark:text-gray-900">
              <Check className="h-3 w-3" />
            </div>
          )}
        </button>
      ))}
    </div>
  );
}
