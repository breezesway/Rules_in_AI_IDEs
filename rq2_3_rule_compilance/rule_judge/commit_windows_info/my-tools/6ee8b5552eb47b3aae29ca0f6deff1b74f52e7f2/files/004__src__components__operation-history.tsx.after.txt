import React, { useState, useEffect } from 'react';
import { History, Clock, FileText, Image, CheckCircle, XCircle, Trash2, FolderOpen } from 'lucide-react';
import { Modal } from '../ui/modal';
import { HistoryRecord } from '../types/electron.d';
import { toast } from 'sonner';

interface OperationHistoryProps {
  isOpen: boolean;
  onClose: () => void;
}

export function OperationHistory({ isOpen, onClose }: OperationHistoryProps) {
  const [history, setHistory] = useState<HistoryRecord[]>([]);
  const [loading, setLoading] = useState(false);

  // 加载历史记录
  const loadHistory = async () => {
    setLoading(true);
    try {
      const records = await window.electronAPI.getOperationHistory();
      setHistory(records);
    } catch (error) {
      console.error('加载历史记录失败:', error);
      toast.error('加载历史记录失败');
    } finally {
      setLoading(false);
    }
  };

  // 清除所有历史记录
  const clearAllHistory = async () => {
    if (!window.confirm('确定要清除所有历史记录吗？此操作不可撤销。')) {
      return;
    }

    try {
      const result = await window.electronAPI.clearOperationHistory();
      if (result.success) {
        setHistory([]);
        toast.success('历史记录已清除');
      } else {
        toast.error('清除失败: ' + result.error);
      }
    } catch (error) {
      console.error('清除历史记录失败:', error);
      toast.error('清除历史记录失败');
    }
  };

  // 删除单条记录
  const deleteRecord = async (id: string) => {
    try {
      const result = await window.electronAPI.deleteHistoryRecord(id);
      if (result.success) {
        setHistory(prev => prev.filter(record => record.id !== id));
        toast.success('记录已删除');
      } else {
        toast.error('删除失败: ' + result.error);
      }
    } catch (error) {
      console.error('删除记录失败:', error);
      toast.error('删除记录失败');
    }
  };

  // 打开文件位置
  const openFileLocation = async (filePath: string) => {
    try {
      await window.electronAPI.openFileLocation(filePath);
    } catch (error) {
      console.error('打开文件位置失败:', error);
      toast.error('打开文件位置失败');
    }
  };

  // 格式化时间
  const formatTime = (timestamp: number) => {
    const date = new Date(timestamp);
    return date.toLocaleString('zh-CN', {
      year: 'numeric',
      month: '2-digit',
      day: '2-digit',
      hour: '2-digit',
      minute: '2-digit',
      second: '2-digit'
    });
  };

  // 格式化文件大小
  const formatFileSize = (bytes: number) => {
    if (bytes === 0) return '0 B';
    const k = 1024;
    const sizes = ['B', 'KB', 'MB', 'GB'];
    const i = Math.floor(Math.log(bytes) / Math.log(k));
    return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
  };

  // 格式化持续时间
  const formatDuration = (ms: number) => {
    if (ms < 1000) return `${ms}ms`;
    if (ms < 60000) return `${(ms / 1000).toFixed(1)}s`;
    return `${Math.floor(ms / 60000)}m ${Math.floor((ms % 60000) / 1000)}s`;
  };

  // 获取操作类型图标
  const getTypeIcon = (type: string) => {
    switch (type) {
      case 'pdf-merge':
        return <FileText className="h-5 w-5 text-red-600" />;
      case 'image-compress':
        return <Image className="h-5 w-5 text-green-600" />;
      default:
        return <FileText className="h-5 w-5 text-gray-600" />;
    }
  };

  // 获取操作类型名称
  const getTypeName = (type: string) => {
    switch (type) {
      case 'pdf-merge':
        return 'PDF合并';
      case 'image-compress':
        return '图片压缩';
      default:
        return '未知操作';
    }
  };

  useEffect(() => {
    if (isOpen) {
      loadHistory();
    }
  }, [isOpen]);

  const footerContent = (
    <div className="flex justify-between">
      <button
        onClick={clearAllHistory}
        disabled={history.length === 0 || loading}
        className="flex items-center space-x-2 px-4 py-3 border border-red-300 text-red-600 rounded-xl hover:bg-red-50 font-medium transition-colors duration-200 disabled:opacity-50 disabled:cursor-not-allowed"
      >
        <Trash2 className="h-4 w-4" />
        <span>清除所有记录</span>
      </button>
      
      <button
        onClick={onClose}
        className="px-6 py-3 border border-gray-300 rounded-xl hover:bg-gray-50 font-medium transition-colors duration-200"
      >
        关闭
      </button>
    </div>
  );

  return (
    <Modal isOpen={isOpen} onClose={onClose} title="操作历史" footer={footerContent}>
      <div className="space-y-4">
        {loading ? (
          <div className="text-center py-8">
            <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-gray-900 mx-auto"></div>
            <p className="mt-2 text-gray-500">加载中...</p>
          </div>
        ) : history.length === 0 ? (
          <div className="text-center py-8">
            <History className="h-12 w-12 text-gray-400 mx-auto mb-4" />
            <p className="text-gray-500">暂无操作记录</p>
          </div>
        ) : (
          <div className="space-y-3 max-h-96 overflow-y-auto">
            {history.map((record) => (
              <div
                key={record.id}
                className={`p-4 rounded-xl border transition-all duration-200 ${
                  record.success 
                    ? 'border-green-200 bg-green-50' 
                    : 'border-red-200 bg-red-50'
                }`}
              >
                {/* 记录头部 */}
                <div className="flex items-center justify-between mb-3">
                  <div className="flex items-center space-x-3">
                    {getTypeIcon(record.type)}
                    <div>
                      <h3 className="font-medium text-gray-900">
                        {getTypeName(record.type)}
                      </h3>
                      <div className="flex items-center space-x-2 text-sm text-gray-500">
                        <Clock className="h-3 w-3" />
                        <span>{formatTime(record.timestamp)}</span>
                        {record.duration && (
                          <>
                            <span>•</span>
                            <span>耗时 {formatDuration(record.duration)}</span>
                          </>
                        )}
                      </div>
                    </div>
                  </div>
                  
                  <div className="flex items-center space-x-2">
                    {record.success ? (
                      <CheckCircle className="h-5 w-5 text-green-600" />
                    ) : (
                      <XCircle className="h-5 w-5 text-red-600" />
                    )}
                    <button
                      onClick={() => deleteRecord(record.id)}
                      className="p-1 hover:bg-gray-200 rounded transition-colors duration-200"
                      title="删除记录"
                    >
                      <Trash2 className="h-4 w-4 text-gray-500" />
                    </button>
                  </div>
                </div>

                {/* 文件信息 */}
                <div className="mb-3">
                  <p className="text-sm font-medium text-gray-700 mb-2">
                    输入文件 ({record.files.length}个):
                  </p>
                  <div className="space-y-1">
                    {record.files.map((file, index) => (
                      <div key={index} className="flex items-center justify-between text-sm">
                        <span className="text-gray-600 truncate">{file.name}</span>
                        <span className="text-gray-500 ml-2">{formatFileSize(file.size)}</span>
                      </div>
                    ))}
                  </div>
                </div>

                {/* 输出信息 */}
                <div className="mb-3">
                  <p className="text-sm font-medium text-gray-700 mb-2">输出文件:</p>
                  <div className="flex items-center justify-between">
                    <div className="flex items-center space-x-2">
                      <span className="text-sm text-gray-600">{record.output.name}</span>
                      {record.output.size && (
                        <span className="text-sm text-gray-500">
                          ({formatFileSize(record.output.size)})
                        </span>
                      )}
                    </div>
                    {record.success && (
                      <button
                        onClick={() => openFileLocation(record.output.path)}
                        className="flex items-center space-x-1 text-sm text-blue-600 hover:text-blue-800"
                        title="打开文件位置"
                      >
                        <FolderOpen className="h-3 w-3" />
                        <span>打开位置</span>
                      </button>
                    )}
                  </div>
                </div>

                {/* 错误信息 */}
                {!record.success && record.error && (
                  <div className="p-2 bg-red-100 border border-red-200 rounded text-sm text-red-700">
                    错误: {record.error}
                  </div>
                )}
              </div>
            ))}
          </div>
        )}
      </div>
    </Modal>
  );
}
